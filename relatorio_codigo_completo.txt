import { useEffect, useRef, useState } from 'react';
import Peer from 'peerjs';
import { v4 as uuidv4 } from 'uuid';
import { Copy, Monitor, Play, Tv, CheckCircle2, Wifi, Laptop2, Smartphone } from 'lucide-react';

function App() {
  const [myPeerId, setMyPeerId] = useState('');
  const [targetPeerId, setTargetPeerId] = useState('');
  const [status, setStatus] = useState('Iniciando sistema...');
  const [isSharing, setIsSharing] = useState(false);
  const [isConnected, setIsConnected] = useState(false);
  const [copied, setCopied] = useState(false);
  
  const videoRef = useRef(null);
  const peerInstance = useRef(null);

  useEffect(() => {
    const newId = uuidv4().substring(0, 5);
    const peer = new Peer(newId);

    peer.on('open', (id) => {
      setMyPeerId(id);
      setStatus('Pronto para conectar');
    });

    peer.on('call', (call) => {
      call.answer();
      call.on('stream', (remoteStream) => {
        if (videoRef.current) {
          videoRef.current.srcObject = remoteStream;
          videoRef.current.play();
          setStatus('Recebendo transmissÃ£o');
          setIsConnected(true);
        }
      });
    });

    peerInstance.current = peer;
    return () => peer.destroy();
  }, []);

  const copyToClipboard = () => {
    navigator.clipboard.writeText(myPeerId);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  const startScreenShare = async () => {
    try {
      const stream = await navigator.mediaDevices.getDisplayMedia({
        video: { cursor: "always" },
        audio: true
      });

      if (videoRef.current) {
        videoRef.current.srcObject = stream;
        videoRef.current.play();
      }

      setIsSharing(true);
      setIsConnected(true);
      setStatus('Compartilhando sua tela');

      stream.getVideoTracks()[0].onended = () => {
        setIsSharing(false);
        setIsConnected(false);
        setStatus('Compartilhamento encerrado');
      };

      peerInstance.current.on('connection', (conn) => { });
      
    } catch (err) {
      console.error("Erro:", err);
      setStatus('Cancelado pelo usuÃ¡rio');
    }
  };

  const connectToHost = () => {
    if (!targetPeerId) return;
    setStatus('Tentando conectar...');
    const conn = peerInstance.current.connect(targetPeerId);
    
    conn.on('open', () => {
        setStatus('Solicitando stream de vÃ­deo...');
        conn.send({ type: 'request-stream', peerId: myPeerId });
    });
  };
  
  useEffect(() => {
     if(!peerInstance.current) return;
     peerInstance.current.on('connection', (conn) => {
        conn.on('data', (data) => {
            if(data.type === 'request-stream' && videoRef.current && videoRef.current.srcObject) {
                peerInstance.current.call(data.peerId, videoRef.current.srcObject);
            }
        });
     });
  }, [isSharing]);

  return (
    // MUDANÃ‡A 1: h-full e overflow-auto permitem rolar se a tela for pequena
    // mas o flex-col organiza tudo.
    <div className="h-full min-h-screen w-full flex flex-col relative overflow-y-auto">
      
      {/* HEADER (Fica no topo, mas nÃ£o "preso" de forma absoluta) */}
      <header className="w-full p-6 flex justify-center shrink-0">
        <div className="w-full max-w-7xl flex items-center justify-between">
            <div className="flex items-center gap-3">
              <div className="bg-gradient-to-br from-blue-600 to-indigo-600 p-2 rounded-lg shadow-lg shadow-blue-500/30">
                <Monitor className="w-5 h-5 text-white" />
              </div>
              <h1 className="text-xl font-bold tracking-tight text-white">
                Share<span className="text-blue-400">Cast</span>
              </h1>
            </div>
            
            {/* Badge de Status */}
            <div className="flex items-center gap-2 bg-slate-800/80 px-3 py-1.5 rounded-full border border-slate-700 backdrop-blur-md">
                <div className={`w-2 h-2 rounded-full ${isConnected ? 'bg-green-500 animate-pulse' : 'bg-blue-500'}`}></div>
                <span className="text-xs font-medium text-slate-300 hidden sm:block">{status}</span>
            </div>
        </div>
      </header>

      {/* MAIN (O segredo da centralizaÃ§Ã£o) */}
      {/* flex-1: Ocupa todo o espaÃ§o vertical disponÃ­vel */}
      {/* flex items-center: Centraliza verticalmente o conteÃºdo interno */}
      <main className="flex-1 w-full flex flex-col items-center justify-center p-4 md:p-8 shrink-0">
        
        <div className="w-full max-w-7xl grid grid-cols-1 lg:grid-cols-12 gap-8">
            
            {/* LADO ESQUERDO: CONTROLES */}
            <div className="lg:col-span-4 flex flex-col gap-5 order-2 lg:order-1 justify-center">
              
              {/* Card Identidade */}
              <div className="bg-slate-900/60 backdrop-blur-xl border border-white/10 rounded-2xl p-6 shadow-2xl">
                <div className="flex items-center gap-2 mb-3 text-slate-400">
                    <Wifi className="w-4 h-4 text-blue-400" />
                    <span className="text-xs font-bold uppercase tracking-widest">Seu ID</span>
                </div>
                <div className="flex gap-2">
                  <input 
                    readOnly 
                    value={myPeerId} 
                    className="w-full h-12 bg-black/40 border border-white/5 rounded-xl text-center font-mono text-xl text-white tracking-widest focus:outline-none focus:ring-2 focus:ring-blue-500/50"
                  />
                  <button 
                    onClick={copyToClipboard}
                    className={`h-12 w-12 flex items-center justify-center rounded-xl border transition-all ${copied ? 'bg-green-500/20 border-green-500 text-green-400' : 'bg-slate-800 border-white/10 hover:bg-slate-700 text-white'}`}
                  >
                    {copied ? <CheckCircle2 size={20} /> : <Copy size={20} />}
                  </button>
                </div>
              </div>

              {/* Card AÃ§Ãµes */}
              <div className="bg-slate-900/60 backdrop-blur-xl border border-white/10 rounded-2xl p-6 shadow-2xl flex flex-col gap-4">
                
                {/* BotÃ£o Host */}
                {!isConnected && (
                    <div>
                        <button 
                            onClick={startScreenShare}
                            className="w-full h-14 bg-blue-600 hover:bg-blue-500 text-white font-bold rounded-xl shadow-lg shadow-blue-600/20 transition-all active:scale-[0.98] flex items-center justify-center gap-3 group"
                        >
                            <div className="bg-white/20 p-1.5 rounded-lg group-hover:scale-110 transition-transform">
                                <Laptop2 size={18} />
                            </div>
                            <span>Compartilhar Tela</span>
                        </button>
                    </div>
                )}

                {!isConnected && (
                     <div className="flex items-center gap-3 py-1">
                        <div className="h-px flex-1 bg-white/10"></div>
                        <span className="text-xs text-slate-500 font-medium">OU ASSISTIR</span>
                        <div className="h-px flex-1 bg-white/10"></div>
                     </div>
                )}

                {/* Input Viewer */}
                <div>
                    <div className="flex gap-2">
                        <input 
                            value={targetPeerId}
                            onChange={(e) => setTargetPeerId(e.target.value)}
                            placeholder="ID do Host..."
                            className="flex-1 h-14 bg-black/40 border border-white/5 rounded-xl px-4 text-white placeholder:text-slate-600 focus:border-green-500/50 focus:outline-none transition-colors"
                        />
                        <button 
                            onClick={connectToHost}
                            className="h-14 w-16 bg-green-600 hover:bg-green-500 text-white rounded-xl shadow-lg shadow-green-600/20 transition-all active:scale-[0.98] flex items-center justify-center"
                        >
                            <Play size={24} fill="currentColor" className="ml-1" />
                        </button>
                    </div>
                    {!isConnected && <p className="text-xs text-slate-500 mt-2 text-center">Cole o ID do computador acima para conectar.</p>}
                </div>
              </div>
            </div>

            {/* LADO DIREITO: VÃDEO */}
            <div className="lg:col-span-8 order-1 lg:order-2">
              <div className="aspect-video w-full bg-black/80 rounded-2xl overflow-hidden border border-white/10 shadow-2xl backdrop-blur-sm relative flex items-center justify-center group">
                
                <video 
                  ref={videoRef} 
                  className={`w-full h-full object-contain transition-opacity duration-700 ${isConnected ? 'opacity-100' : 'opacity-0'}`}
                  autoPlay
                  playsInline
                  muted
                />

                {!isConnected && (
                  <div className="absolute inset-0 flex flex-col items-center justify-center p-6 text-center">
                    <div className="w-20 h-20 bg-white/5 rounded-full flex items-center justify-center mb-4 ring-1 ring-white/10 animate-pulse">
                        <Monitor className="w-8 h-8 text-slate-500" />
                    </div>
                    <h3 className="text-lg font-medium text-slate-300">Nenhum sinal de vÃ­deo</h3>
                  </div>
                )}

                {/* Status Overlay */}
                {isConnected && (
                    <div className="absolute top-4 left-4 bg-black/60 backdrop-blur px-3 py-1 rounded-lg border border-white/10 flex items-center gap-2">
                         <span className="w-2 h-2 rounded-full bg-red-500 animate-pulse"></span>
                         <span className="text-xs font-bold text-white tracking-wider">AO VIVO</span>
                    </div>
                )}
              </div>
            </div>

        </div>
      </main>

      {/* FOOTER */}
      <footer className="p-6 text-center text-slate-600 text-xs shrink-0">
        &copy; 2025 ShareCast P2P
      </footer>

    </div>
  );
}

export default App;
